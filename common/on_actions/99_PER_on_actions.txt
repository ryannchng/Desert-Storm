on_actions = {
	on_weekly_PER = {
		effect = {
			if = {
				limit = {
					has_country_flag = PER_protests_ongoing
					check_variable = {
						PER_protest_exhaustion > 19
					}
				}
				random = {
					chance = 30
					country_event = iranian_events.202
				}
			}
			if = {
				limit = {
					has_country_flag = PER_protests_ongoing
				}
				if = { # Protests are not tired
					limit = {
						check_variable = {
							PER_protest_exhaustion < 10
						}
					}
					random = {
						chance = 70
						PER_trigger_protest = yes
					}
				}
				else_if = { # Protests are tired
					limit = {
						check_variable = {
							PER_protest_exhaustion > 15
						}
					}
					random = {
						chance = 20
						PER_trigger_protest = yes
					}
				}
				else_if = { # Protests are starting to tire
					limit = {
						check_variable = {
							PER_protest_exhaustion > 9
						}
					}
					random = {
						chance = 40
						PER_trigger_protest = yes
					}
				}
			}
		}
	}
	on_monthly_PER = {
		effect = {
			# Neutral Oriented Foreign Policy
			if = {
				limit = {
					has_idea = PER_neutral_political_scene
				}
				random = {
					chance = 50
					PER_neutral_politics = yes
				}
			}

			# Remove-candidate Iraqi insurgent and Palestinian proxy hooks pruned for Desert Storm scope

			# For Election
			if = {
				limit = {
					NOT = {
						has_country_flag = PER_majles_abolished
					}
					check_variable = {
						PER_election_var > 0
					}
				}
				add_to_variable = { var = PER_election_var value = -1 }
				clamp_variable = { var = PER_election_var min = 0 max = 48 }
			}
			if = {
				limit = {
					NOT = {
						has_country_flag = PER_majles_abolished
					}
					check_variable = {
						PER_election_var = 0
					}
				}
				add_to_variable = { var = PER_election_var value = 48 }
				clamp_variable = { var = PER_election_var min = 0 max = 48 }
				country_event = {
					id = PER_majlis.1
					days = 18
				}
			}
			if = {
				limit = {
					NOT = {
						has_country_flag = PER_majles_abolished
					}
					check_variable = {
						PER_election_var > 0
					}
				}
				add_to_variable = { var = PER_session_var value = -1 }
				clamp_variable = { var = PER_session_var min = 0 max = 48 }
			}
			if = {
				limit = {
					NOT = {
						has_country_flag = PER_majles_abolished
					}
					check_variable = {
						PER_session_var = 0
					}
				}
				add_to_variable = { var = PER_session_var value = 18 }
				clamp_variable = { var = PER_session_var min = 0 max = 48 }
				country_event = {
					id = PER_majlis.2
				}
			}
			# Jordan Shale Oil
			if = {
				limit = {
					JOR = {
						has_country_flag = JOR_agreed_oil_shale
					}
					check_variable = {
						PER_shale_contract < 120
					}
				}
				add_to_variable = { var = PER_shale_contract value = 1 }
				clamp_variable = { var = PER_shale_contract min = 0 max = 120 }
			}
			if = {
				limit = {
					JOR = {
						has_country_flag = JOR_agreed_oil_shale
					}
					check_variable = {
						PER_shale_contract = 120
					}
				}
				PER = {
					country_event = iranian_focus.17
				}
				JOR = {
					country_event = iranian_focus.17
				}
			}
			# Basij Fervor stuff
			if = {
				limit = {
					has_country_flag = PER_basij_fervor_decisions
					NOT = {
						has_country_flag = PER_basij_variable_stop
					}
				}
				add_to_variable = { var = islamic_propaganda_status value = 5 }
				clamp_variable = { var = islamic_propaganda_status min = 0 max = 100 }
			}
			if = {
				limit = {
					has_country_flag = PER_basij_fervor_decisions
					has_country_flag = PER_basij_fervor_drain
				}
				add_to_variable = { var = islamic_propaganda_status value = -10 }
				clamp_variable = { var = islamic_propaganda_status min = 0 max = 100 }
			}
		}
	}
}
