clamp_shia = {
	clamp_variable = {
		var = shia_population
		min = 0
		max = 100
	}
	if = {
		limit = {
			NOT = {
				has_idea = tiny_shia_population
				has_idea = small_shia_population
				has_idea = substantial_shia_population
				has_idea = large_shia_population
			}
			check_variable = { shia_population > 0 }
		}
		add_ideas = { tiny_shia_population }
	}
	if = {
		limit = {
			has_idea = tiny_shia_population
			check_variable = { shia_population = 0 }
		}
		remove_ideas = { tiny_shia_population }
	}
	if = {
		limit = {
			has_idea = tiny_shia_population
			check_variable = { shia_population > 10 }
		}
		swap_ideas = {
			remove_idea = tiny_shia_population
			add_idea = small_shia_population
		}
	}
	if = {
		limit = {
			has_idea = small_shia_population
			check_variable = { shia_population < 11 }
		}
		swap_ideas = {
			remove_idea = small_shia_population
			add_idea = tiny_shia_population
		}
	}
	if = {
		limit = {
			has_idea = small_shia_population
			check_variable = { shia_population > 19 }
		}
		swap_ideas = {
			remove_idea = small_shia_population
			add_idea = substantial_shia_population
		}
	}
	if = {
		limit = {
			has_idea = substantial_shia_population
			check_variable = { shia_population < 20 }
		}
		swap_ideas = {
			remove_idea = substantial_shia_population
			add_idea = small_shia_population
		}
	}
	if = {
		limit = {
			has_idea = substantial_shia_population
			check_variable = { shia_population > 49 }
		}
		swap_ideas = {
			remove_idea = substantial_shia_population
			add_idea = large_shia_population
		}
	}
	if = {
		limit = {
			has_idea = large_shia_population
			check_variable = { shia_population < 50 }
		}
		swap_ideas = {
			remove_idea = large_shia_population
			add_idea = substantial_shia_population
		}
	}
}

# Effect: calculate_monthly_arab_spring_change
# Purpose: Calculate the change in protest strength for the month
calculate_monthly_arab_spring_change = {
	set_variable = { arab_spring_change = 0 }
	# Police Brutality Increases Arab Spring Likelihood
	if = { limit = { has_idea = police_05 }
		add_to_variable = { arab_spring_change = 3 }
	}
	else_if = {
		limit = { has_idea = police_04 }
		add_to_variable = { arab_spring_change = 2 }
	}
	else_if = { limit = { has_idea = police_03 }
		add_to_variable = { arab_spring_change = 1 }
	}

	# Economic Situation Increases Arab Spring Likelihood
	if = {
		limit = { check_variable = { gdp_per_capita < 15 } }
		if = { limit = { has_idea = depression }
			add_to_variable = { arab_spring_change = 6 }
		}
		else_if = { limit = { has_idea = recession }
			add_to_variable = { arab_spring_change = 4 }
		}
		else_if = { limit = { has_idea = stagnation }
			add_to_variable = { arab_spring_change = 2 }
		}
	}

	# Social Unrest Increases Arab Spring Likelihood
	if = {
		limit = { has_idea = social_01 }
		add_to_variable = { arab_spring_change = 3 }
	}
	else_if = {
		limit = { has_idea = social_02 }
		add_to_variable = { arab_spring_change = 2 }
	}
	else_if = {
		limit = { has_idea = social_03 }
		add_to_variable = { arab_spring_change = 1 }
	}
	else_if = {
		limit = { has_idea = social_04 }
		add_to_variable = { arab_spring_change = -1 }
	}
	else_if = {
		limit = { has_idea = social_05 }
		add_to_variable = { arab_spring_change = -2 }
	}
	else_if = {
		limit = { has_idea = social_06 }
		add_to_variable = { arab_spring_change = -3 }
	}

	# Healthcare Spending
	if = {
		limit = { has_idea = health_04 }
		add_to_variable = { arab_spring_change = -1 }
	}
	else_if = {
		limit = { has_idea = health_05 }
		add_to_variable = { arab_spring_change = -2 }
	}
	else_if = {
		limit = { has_idea = health_06 }
		add_to_variable = { arab_spring_change = -3 }
	}

	# Corruption Increases Arab Spring Likelihood
	if = { limit = { has_idea = paralyzing_corruption }
		add_to_variable = { arab_spring_change = 6 }
	}
	else_if = { limit = { has_idea = crippling_corruption }
		add_to_variable = { arab_spring_change = 5 }
	}
	else_if = { limit = { has_idea = rampant_corruption }
		add_to_variable = { arab_spring_change = 4 }
	}
	else_if = { limit = { has_idea = unrestrained_corruption }
		add_to_variable = { arab_spring_change = 3 }
	}
	else_if = { limit = { has_idea = systematic_corruption }
		add_to_variable = { arab_spring_change = 2 }
	}
	else_if = { limit = { has_idea = widespread_corruption }
		add_to_variable = { arab_spring_change = 1 }
	}
	else_if = { limit = { has_idea = modest_corruption }
		add_to_variable = { arab_spring_change = -1 }
	}
	else_if = { limit = { has_idea = slight_corruption }
		add_to_variable = { arab_spring_change = -2 }
	}
	else_if = { limit = { has_idea = negligible_corruption }
		add_to_variable = { arab_spring_change = -3 }
	}

	# Historical Section
	if = { limit = { is_historical_focus_on = yes }
		if = {
			# DS_REVIEW: Arab Spring logic retained for SYR/EGY — evaluate for 1990 rework
			limit = {
				original_tag = SYR
				is_in_array = { ruling_party = 7 }
			}
			add_to_variable = { arab_spring_change = 2 }
		}
		else_if = {
			# DS_REVIEW: Arab Spring logic retained for SYR/EGY — evaluate for 1990 rework
			limit = {
				original_tag = EGY
				is_in_array = { ruling_party = 0 }
			}
			add_to_variable = { arab_spring_change = 5 }
		}
	}

	# Shia Resistance Increases Arab Spring Likelihood
	if = {
		limit = {
			OR = {
				has_idea = shia_resistance_05
				has_idea = shia_resistance_06
				has_idea = shia_resistance_07
				has_idea = shia_resistance_08
				has_idea = shia_resistance_09
				has_idea = shia_resistance_10
			}
		}
		add_to_variable = { arab_spring_change = 2 }
	}
	if = {
		limit = {
			OR = {
				AND = {
					has_elections = yes
					OR = {
						has_government = democratic
						has_government = neutrality
					}
					NOT = {
						has_idea = hybrid_regime_emerging
						has_idea = hybrid_regime_western
					}
				}
				has_idea = elected_legislature
			}
		}
		subtract_from_variable = { arab_spring_change = 4 }
	}
	if = {
		limit = {
			NOT = { has_tech = internet1 }
		}
		subtract_from_variable = { arab_spring_change = 4 }
	}

	if = {
		limit = {
			check_variable = { government_coalition_strength <  0.500 }
		}
		add_to_variable = { arab_spring_change = 2 }
	}
	else_if = {
		limit = {
			check_variable = { government_coalition_strength > 0.499 }
			check_variable = { government_coalition_strength < 0.599 }
		}
		add_to_variable = { arab_spring_change = -1 }
	}
	else_if = {
		limit = {
			check_variable = { government_coalition_strength > 0.599 }
			check_variable = { government_coalition_strength < 0.699 }
		}
		add_to_variable = { arab_spring_change = -2 }
	}
	else_if = {
		limit = {
			check_variable = { government_coalition_strength > 0.699 }
			check_variable = { government_coalition_strength < 0.799 }
		}
		add_to_variable = { arab_spring_change = -3 }
	}
	else_if = {
		limit = {
			check_variable = { government_coalition_strength > 0.799 }
			check_variable = { government_coalition_strength < 0.899 }
		}
		add_to_variable = { arab_spring_change = -4 }
	}
	else_if = {
		limit = {
			check_variable = { government_coalition_strength > 0.899 }
		}
		add_to_variable = { arab_spring_change = -5 }
	}
}

# Effect: increment_arab_spring_strength
# Purpose: Increment the strength of the Arab Spring protests
increment_arab_spring_strength = {
	calculate_monthly_arab_spring_change = yes
	set_temp_variable = { temp_strength = arab_spring_change }
	change_arab_spring_strength = yes
}


# Effect: Update social conservatism idea
# Purpose: Updates the social conservatism idea for Arab nations
update_social_conservatism_idea = {
	if = {
		limit = { check_variable = { social_conservatism_society > 90 } }
		if = {
			limit = { has_idea = conservative_society }
			swap_ideas = {
				remove_idea = conservative_society
				add_idea = extremely_conservative_society
			}
		}
		else_if = {
			limit = { has_idea = shia_conservative_society }
			swap_ideas = {
				remove_idea = shia_conservative_society
				add_idea = shia_extremely_conservative_society
			}
		}
	}
	else_if = {
		limit = { check_variable = { social_conservatism_society > 64 } }
		if = {
			limit = { has_idea = extremely_conservative_society }
			swap_ideas = {
				remove_idea = extremely_conservative_society
				add_idea = conservative_society
			}
		}
		else_if = {
			limit = { has_idea = shia_extremely_conservative_society }
			swap_ideas = {
				remove_idea = shia_extremely_conservative_society
				add_idea = shia_conservative_society
			}
		}
	}
	else_if = {
		limit = { check_variable = { social_conservatism_society > 49 } }
		if = {
			limit = { has_idea = conservative_society }
			swap_ideas = {
				remove_idea = conservative_society
				add_idea = moderately_conservative_society
			}
		}
		else_if = {
			limit = { has_idea = shia_conservative_society }
			swap_ideas = {
				remove_idea = shia_conservative_society
				add_idea = shia_moderately_conservative_society
			}
		}
	}
	else_if = {
		limit = { check_variable = { social_conservatism_society > 29 } }
		if = {
			limit = { has_idea = moderately_conservative_society }
			swap_ideas = {
				remove_idea = moderately_conservative_society
				add_idea = moderately_liberal_society
			}
		}
		else_if = {
			limit = { has_idea = shia_moderately_conservative_society }
			swap_ideas = {
				remove_idea = shia_moderately_conservative_society
				add_idea = shia_moderately_liberal_society
			}
		}
	}
	else_if = {
		limit = { check_variable = { social_conservatism_society > 9 } }
		if = {
			limit = { has_idea = moderately_liberal_society }
			swap_ideas = {
				remove_idea = moderately_liberal_society
				add_idea = liberal_society
			}
		}
		else_if = {
			limit = { has_idea = shia_moderately_liberal_society }
			swap_ideas = {
				remove_idea = shia_moderately_liberal_society
				add_idea = shia_liberal_society
			}
		}
	}
	else = {
		if = {
			limit = { has_idea = liberal_society }
			swap_ideas = {
				remove_idea = liberal_society
				add_idea = extremely_liberal_society
			}
		}
		else_if = {
			limit = { has_idea = shia_liberal_society }
			swap_ideas = {
				remove_idea = shia_liberal_society
				add_idea = shia_extremely_liberal_society
			}
		}
	}
}

# On Actions
on_weekly_gcc_dominant_member_check = {
	if = { limit = { has_country_flag = gcc_dominant_member }
		# Gulf Cooperation Council Unity Additions
		if = { limit = { threat > 0.75 }
			add_to_variable = { global.gcc_unity = 0.75 }
		}
		else_if = { limit = { threat > 0.50 }
			add_to_variable = { global.gcc_unity = 0.50 }
		}
		else_if = { limit = { threat > 0.25 }
			add_to_variable = { global.gcc_unity = 0.25 }
		}
		clamp_variable = { var = global.gcc_unity min = 0 max = 100 }

		if = {
			limit = {
				has_idea = idea_gcc_member_state
				has_completed_focus = GCC_limited_unified_command
				NOT = { has_completed_focus = GCC_a_robust_alliance }
				is_in_faction = no
				has_defensive_war = yes
				all_enemy_country = {
					NOT = { has_idea = idea_gcc_member_state }
				}
			}
			create_faction_from_template = faction_template_gcc
			# Add to Faction
			for_each_scope_loop = {
				array = global.gcc_member_state
				PREV = { add_to_faction = PREV }
			}
			set_country_flag = gcc_formed
		}
		else_if = {
			limit = {
				NOT = {
					any_of_scopes = {
						array = global.gcc_member_state
						OR = {
							has_war = yes
							has_completed_focus = GCC_a_robust_alliance
						}
					}
				}
				any_of_scopes = {
					array = global.gcc_member_state
					has_war = no
					has_country_flag = gcc_formed
				}
			}
			if = { limit = { is_faction_leader = yes }
				dismantle_faction = yes
			}
			# Add to Faction
			for_each_scope_loop = {
				array = global.gcc_member_state
				if = { limit = { has_completed_focus = GCC_joint_defense_agreement }
					for_each_scope_loop = {
						array = global.gcc_member_state
						if = { limit = { NOT = { tag = PREV } }
							give_guarantee = PREV
						}
					}
				}
			}
		}
	}
}
